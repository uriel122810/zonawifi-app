# This file contains error messages which are shown to user, when http/https
# login is used.
# These messages can be changed to make user interface more friendly, including
# translations to different languages.
#
# Various variables can be used here as well. Most frequently used ones are:
#	$(error-orig)	- original error message from hotspot
#	$(ip)		- ip address of a client
#	$(username)	- username of client trying to log in

# error interno
# Nunca debería suceder. Si es así, se mostrará la página de error.
# mostrando este mensaje de error (error-orig describirá lo que sucedió)

internal-error = error internal ($(error-orig))

# error de configuración
# Nunca debería suceder si el punto de acceso está configurado correctamente.

config-error = errores de configuración ($(error-orig))

# Sin iniciar sesión
# Sucederá, si el usuario solicita la página de estado o de cierre de sesión,
# que en realidad no está conectado

not-logged-in = no estas en (ip $(ip))

# ippool-vacío
# La dirección IP para el usuario debe asignarse desde el grupo de ip, pero no hay más
# direcciones en ese grupo

ippool-empty = no se puede asignar la dirección IP - no más direcciones libres del grupo

# shutting-down
# When shutdown is executed, new clients are not accepted

shutting-down = cubrir el servicio de punto de acceso

# user-session-limit
# If user profile has limit of shared-users, then this error will be shown
# after reaching this limit

user-session-limit = el cupón/código de usuario está activo $(username)

# license-session-limit
# Depending on licence number of active hotspot clients is limited to
# one or another amount. If this limit is reached, following error is displayed.

license-session-limit = La duración de su cupón ha expirado ($(error-orig))

# nombre-de-usuario-mac-incorrecto
# Si el nombre de usuario se parece a la dirección MAC (12:34:56:78:9a:bc), pero no lo es
# una dirección MAC de este cliente, se rechaza el inicio de sesión

wrong-mac-username = el vale/código de usuario no coincide ($(username)): MAC address No es tuyo

# cap-desaparecido
# Si se usa el método de inicio de sesión http-chap, pero el programa de punto de acceso no recibe
# Contraseña encriptada, se muestra este mensaje de error.
# Posibles razones de falla:
# - JavaScript no está habilitado en el navegador web;
# - la página login.html no es válida;
# - el valor del desafío ha expirado en el servidor (más de 1 hora de inactividad);
# - El método de inicio de sesión http-chap se eliminó recientemente;
# Si JavaScript está habilitado y la página login.html es válida,
# luego, volver a intentar iniciar sesión generalmente soluciona este problema.

chap-missing = navegador web no envió respuesta de desafío (try again, enable JavaScript)

# invalid-username
# Most general case of invalid username or password. If RADIUS server
# has sent an error string with Access-Reject message, then it will
# override this setting.

invalid-username = El cupón/código de usuario no coincide, vuelva a ingresarlo correctamente.

# invalid-mac
# Local users (on hotspot server) can be bound to some MAC address. If login
# from different MAC is tried, this error message will be shown.

invalid-mac = el cupón/código de usuario $(username)) no se puede activar desde esta dirección MAC

# uptime-limit, traffic-limit
# For local hotspot users in case if limits are reached

uptime-limit = el cupón/código de usuario $(username) ha caducado
traffic-limit = código de cupón/usuario $(username) ha alcanzado el límite de cuota

# radius-timeout
# User is authenticated by RADIUS server, but no response is received from it,
# following error will be shown.

radius-timeout = RADIUS el servidor no responde

# autenticación en curso
# Autorización en curso. El cliente ya ha emitido una solicitud de autorización
# que aún no está completo.

auth-in-progress = ya está habilitado, inténtalo de nuevo más tarde

# radius-reply
# Radius server returned some custom error message

radius-reply = $(error-orig)
